// LyricLive Builder: plays the show in sync for everyone, late joiners too, and turns each gaze guide
// toward where that viewer really looks (design §10.7). The builder copies this file to LLBSync.cs
// only when UdonSharp is in the project (as a .txt it is never compiled).
//
// Sync: pressing the button (Interact) makes the presser the owner, who stores the server time of
// "t = 0"; every client seeks the director to (now - start). Udon cannot change the playback speed,
// so drift is corrected by re-seeking only when it exceeds maxDrift (prototype S0-11).
// Gaze: when a guide starts, the head direction is read. Already looking at the line: the guide is
// hidden. Otherwise the motes wait where the viewer looks, then flow from there to the line, taking
// longer the farther it is. Timeline keeps their visibility, colour and size; LateUpdate runs after it
// and replaces their baked positions. Visual only, nothing is synced.
using UdonSharp;
using UnityEngine;
using UnityEngine.Playables;
using VRC.SDKBase;

[UdonBehaviourSyncMode(BehaviourSyncMode.Manual)]
public class LLBSync : UdonSharpBehaviour
{
    public PlayableDirector director;
    public Transform stage;
    public float maxDrift = 0.2f;
    public float guideAngle = 30f;
    public Transform[] guidePivots;
    public float[] guideOn;
    public float[] guideHit;
    public Vector3[] guideTarget;                  // stage space
    public int[] guideMotes;

    [UdonSynced] private double startServerTime;
    [UdonSynced] private bool playing;
    private int[] guideState;                      // 0 waiting, 1 shown or hidden for this viewer, 2 done
    private Vector3[] guideGaze;                   // the viewer's gaze when the guide began
    private float[] guideStart;                    // when the streak leaves it
    private Vector3 headPosition, headForward;

    // the same values as core/guide.py
    private const float LeadMin = 0.7f, LeadMax = 1.2f, MoteLag = 0.03f, Bow = 0.14f;
    private const float MinDistance = 1.5f;        // a streak never starts closer to the face than this

    private void Start()
    {
        int n = guidePivots == null ? 0 : guidePivots.Length;
        guideState = new int[n];
        guideGaze = new Vector3[n];
        guideStart = new float[n];
    }

    public override void Interact()
    {
        if (!Networking.IsOwner(gameObject)) Networking.SetOwner(Networking.LocalPlayer, gameObject);
        playing = !playing;
        startServerTime = Networking.GetServerTimeInSeconds();
        RequestSerialization();
        Apply();
    }

    public override void OnDeserialization()
    {
        Apply();
    }

    private void Apply()
    {
        if (director == null) return;
        ResetGuides();
        if (!playing)
        {
            director.Stop();
            return;
        }
        double t = Networking.GetServerTimeInSeconds() - startServerTime;
        if (t < 0) t = 0;
        if (t > director.duration) return;
        director.time = t;
        director.Play();
    }

    private void Update()
    {
        if (!playing || director == null) return;
        double t = Networking.GetServerTimeInSeconds() - startServerTime;
        if (t > director.duration) return;
        if (System.Math.Abs(director.time - t) > maxDrift)
        {
            director.time = t;
            director.Evaluate();
        }
        Guides((float)director.time);
    }

    private void LateUpdate()
    {
        if (!playing || director == null || guideState == null) return;
        float t = (float)director.time;
        for (int k = 0; k < guideState.Length; k++)
            if (guideState[k] == 1 && guidePivots[k] != null && guidePivots[k].gameObject.activeSelf) MoveMotes(k, t);
    }

    private void ReadHead()
    {
        VRCPlayerApi me = Networking.LocalPlayer;
        if (Utilities.IsValid(me))
        {
            VRCPlayerApi.TrackingData head = me.GetTrackingData(VRCPlayerApi.TrackingDataType.Head);
            headPosition = head.position;
            headForward = head.rotation * Vector3.forward;
        }
        else
        {
            headPosition = stage.position + Vector3.up * 1.5f;
            headForward = stage.forward;
        }
    }

    private Vector3 Arc(Vector3 from, Vector3 to, float s)
    {
        if (Vector3.Angle(from, to) < 170f) return Vector3.Slerp(from, to, s);
        Vector3 over = Vector3.ProjectOnPlane(Vector3.up, from).normalized;   // straight behind: over the top
        return s < 0.5f ? Vector3.Slerp(from, over, s * 2f) : Vector3.Slerp(over, to, s * 2f - 1f);
    }

    private void MoveMotes(int k, float t)
    {
        ReadHead();
        Transform pivot = guidePivots[k];
        Vector3 line = stage.TransformPoint(guideTarget[k]) - headPosition;
        float distance = line.magnitude;
        Vector3 to = line.normalized;
        for (int m = 0; m < guideMotes[k]; m++)
        {
            float t0 = guideStart[k] + m * MoteLag;
            float s = Mathf.Clamp01((t - t0) / (guideHit[k] + m * MoteLag - t0));
            s = 0.5f - 0.5f * Mathf.Cos(Mathf.PI * s);
            Vector3 dir = (Arc(guideGaze[k], to, s) + Vector3.up * (Mathf.Sin(Mathf.PI * s) * Bow)).normalized;
            pivot.GetChild(m).position = headPosition + dir * Mathf.Lerp(Mathf.Max(distance, MinDistance), distance, s);
        }
    }

    private void Guides(float t)
    {
        if (guideState == null) return;
        for (int k = 0; k < guideState.Length; k++)
        {
            Transform pivot = guidePivots[k];
            if (pivot == null) continue;
            if (guideState[k] == 0 && t >= guideOn[k] - 0.05f && t < guideHit[k])
            {
                guideState[k] = 1;
                ReadHead();
                float angle = Vector3.Angle(headForward, stage.TransformPoint(guideTarget[k]) - headPosition);
                if (angle < guideAngle)
                {
                    pivot.gameObject.SetActive(false);   // already looking there: no guide
                }
                else
                {
                    guideGaze[k] = headForward;
                    guideStart[k] = guideHit[k] - Mathf.Lerp(LeadMin, LeadMax, angle / 180f);
                }
            }
            else if (guideState[k] == 1 && t > guideHit[k] + 1.0f)
            {
                guideState[k] = 2;
                pivot.gameObject.SetActive(true);
            }
        }
    }

    private void ResetGuides()
    {
        if (guideState == null) return;
        for (int k = 0; k < guideState.Length; k++)
        {
            guideState[k] = 0;
            if (guidePivots[k] != null) guidePivots[k].gameObject.SetActive(true);
        }
    }
}
